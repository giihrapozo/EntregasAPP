EXERCÍCIO 1 — STACK NAVIGATION (1.a, 1.b e 1.c)


import React from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';

const Stack = createNativeStackNavigator();

function TelaInicial({ navigation }) {
  return (
    <View style={estilos.tela}>
      <Text style={estilos.titulo}>Olá, navegação!</Text>

      <TouchableOpacity
        style={estilos.botao}
        onPress={() => navigation.navigate('Detalhes')}
      >
        <Text style={estilos.textoBotao}>Ir para Detalhes</Text>
      </TouchableOpacity>
    </View>
  );
}

function TelaDetalhes({ navigation }) {
  return (
    <View style={estilos.tela}>
      <Text style={estilos.titulo}>Estou na tela de detalhes</Text>

      <TouchableOpacity
        style={estilos.botao}
        onPress={() => navigation.goBack()}
      >
        <Text style={estilos.textoBotao}>Voltar</Text>
      </TouchableOpacity>
    </View>
  );
}

export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator initialRouteName="Inicio">
        <Stack.Screen
          name="Inicio"
          component={TelaInicial}
          options={{ title: 'Home' }}
        />

        <Stack.Screen
          name="Detalhes"
          component={TelaDetalhes}
          options={{ title: 'Detalhes do Item' }}
        />
      </Stack.Navigator>
    </NavigationContainer>
  );
}

const estilos = StyleSheet.create({
  tela: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
  },
  titulo: {
    fontSize: 24,
    fontWeight: 'bold',
    marginBottom: 20,
  },
  botao: {
    backgroundColor: '#007AFF',
    padding: 14,
    borderRadius: 8,
  },
  textoBotao: {
    color: '#fff',
    fontSize: 16,
  },
});

import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  FlatList,
  Button,
  StyleSheet
} from 'react-native';

import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';

const Stack = createNativeStackNavigator();

const produtos = [
  { id: '1', nome: 'Notebook', preco: 3500, descricao: 'Notebook 16GB RAM' },
  { id: '2', nome: 'Smartphone', preco: 2200, descricao: 'Tela 6.5", 128GB' },
  { id: '3', nome: 'Fone Bluetooth', preco: 350, descricao: 'Cancelamento de ruído' },
  { id: '4', nome: 'Teclado', preco: 150, descricao: 'Teclado sem fio' },

function TelaInicial({ navigation }) {
  return (
    <View style={estilos.container}>
      <Text style={estilos.titulo}>Exercício 2</Text>

      <Button title="2.a - Enviar nome" onPress={() => navigation.navigate('EnviarNome')} />
      <Button title="2.b - Cadastro" onPress={() => navigation.navigate('Cadastro')} />

// QUESTÃO 2.a — Enviar um parâmetro simples

function TelaEnviarNome({ navigation }) {
  const [nomeDigitado, setNomeDigitado] = useState('');

  return (
    <View style={estilos.container}>
      <TextInput
        style={estilos.input}
        placeholder="Digite seu nome"
        value={nomeDigitado}
        onChangeText={setNomeDigitado}
      />

      <Button
        title="Enviar Nome"
        onPress={() => navigation.navigate('Detalhes', { nome: nomeDigitado })}
      />
    </View>
  );
}

// QUESTÃO 2.b — Enviar vários parâmetros

function TelaCadastro({ navigation }) {
  const [nome, setNome] = useState('');
  const [email, setEmail] = useState('');
  const [idade, setIdade] = useState('');

  return (
    <View style={estilos.container}>
      <TextInput style={estilos.input} placeholder="Nome" value={nome} onChangeText={setNome} />
      <TextInput style={estilos.input} placeholder="Email" value={email} onChangeText={setEmail} />
      <TextInput style={estilos.input} placeholder="Idade" value={idade} onChangeText={setIdade} />

      <Button
        title="Cadastrar"
        onPress={() => navigation.navigate('DetalhesCadastro', { nome, email, idade })}
      />
    </View>
  );
}

function TelaDetalhesCadastro({ route, navigation }) {
  const { nome, email, idade } = route.params;

  return (
    <View style={estilos.container}>
      <Text>Nome: {nome}</Text>
      <Text>Email: {email}</Text>
      <Text>Idade: {idade}</Text>

// Tela de detalhes usada na questão 2.a

function TelaDetalhes({ route, navigation }) {

// QUESTÃO 2.c — FlatList de produtos

function TelaLista({ navigation }) {

  );

  return (
    <NavigationContainer>
      <Tab.Navigator
        screenOptions={{
          tabBarActiveTintColor: '#28a745',
          tabBarInactiveTintColor: '#999',
        }}
      >
        <Tab.Screen
          name="Home"
          component={TelaHome}
          options={{
            tabBarLabel: 'Início',
            tabBarIcon: () => <Text>H</Text>,
